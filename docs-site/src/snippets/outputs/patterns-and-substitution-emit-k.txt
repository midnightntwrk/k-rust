module KRS-FRAMEWORK
  syntax KItem
  syntax KResult

  syntax KItem ::= KResult
endmodule

module NAT-SYNTAX
  imports KRS-FRAMEWORK

  syntax Answer
  syntax Nat

  syntax Answer ::= "no" [symbol(answerSyntaxNo), format(%1)]
  syntax Answer ::= "yes" [symbol(answerSyntaxYes), format(%1)]
  // `isZero(n)` answers whether `n` is `zero`.
  syntax Answer ::= "isZero" "(" Nat ")" [symbol(isZero), function, total]
  syntax Nat ::= "s" "(" Nat ")" [symbol(natSyntaxSucc), format(%1%2%3%4)]
  syntax Nat ::= "zero" [symbol(natSyntaxZero), format(%1)]
  // `plus(m, n)` is the sum of `m` and `n`.
  syntax Nat ::= "plus" "(" Nat "," Nat ")" [symbol(plus), function, total]
  // `same(m, n)` is `yes` when `m` and `n` are the same number.
  syntax Answer ::= "same" "(" Nat "," Nat ")" [symbol(same), function]
endmodule

module NAT
  imports KRS-FRAMEWORK
  imports NAT-SYNTAX

  syntax KItem ::= Nat
  syntax KItem ::= Answer

  configuration
    <k> $PGM:Answer </k>


  rule isZero ( s ( _ ) ) => no [label(nat--is_zero_succ)]
  rule isZero ( zero ) => yes [label(nat--is_zero_zero)]
  rule plus ( s ( M:Nat ) , N:Nat ) => s ( plus ( M:Nat , N:Nat ) ) [label(nat--plus_succ)]
  rule plus ( zero , N:Nat ) => N:Nat [label(nat--plus_zero)]
  rule same ( N:Nat , N:Nat ) => yes [label(nat--same_equal)]
endmodule
