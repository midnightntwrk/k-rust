module KRS-FRAMEWORK
  syntax KItem
  syntax KResult

  syntax KItem ::= KResult
endmodule

module NAT-SYNTAX
  imports KRS-FRAMEWORK

  syntax Nat

  syntax Nat ::= "s" "(" Nat ")" [symbol(natSyntaxSucc), format(%1%2%3%4)]
  syntax Nat ::= "zero" [symbol(natSyntaxZero), format(%1)]
  // `plus(m, n)` is the sum of `m` and `n`.
  syntax Nat ::= "plus" "(" Nat "," Nat ")" [symbol(plus), function, total]
  // `pred(n)` is the number before `n`; no equation constrains `pred(zero)`.
  syntax Nat ::= "pred" "(" Nat ")" [symbol(pred), function]
  // `times(m, n)` is the product of `m` and `n`.
  syntax Nat ::= "times" "(" Nat "," Nat ")" [symbol(times), function, total]
endmodule

module NAT
  imports KRS-FRAMEWORK
  imports NAT-SYNTAX
